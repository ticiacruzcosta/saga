// Motor de Integridade Acadêmica — Validação em Tempo Real (RN01, RN02, RN03)
export function validarOferta(
  turmas: TurmaParaValidar[],
  rodizios: Array<[number, number]>
): Map<number, Conflito[]> {
  const conflitos = new Map<number, Conflito[]>();

  // RN01: Restrição rigorosa de choque docente no mesmo dia e faixa horária
  for (const [chaveSlot, alocadas] of agruparPorDiaHorario(turmas)) {
    verificarChoqueDocente(alocadas, conflitos);
    verificarColisaoAmbiente(alocadas, rodizios, conflitos);
  }

  // RN02 & RN03: Capacidade física do laboratório e travas (Lock)
  for (const t of turmas) {
    if (t.ambienteCapacidade && t.vagas > t.ambienteCapacidade) {
      registrarErro(conflitos, t.id, `Lotação (${t.vagas}) excede teto (${t.ambienteCapacidade})`);
    }
  }
  return conflitos;
}
